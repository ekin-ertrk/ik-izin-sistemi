            '2026-01-01', '2026-04-23', '2026-05-01', '2026-05-19', '2026-07-15', '2026-08-30', '2026-10-29',

        const isGunuHesapla = (baslangicStr, bitisStr) => {

            if (!baslangicStr || !bitisStr) return 0;

            let baslangic = new Date(baslangicStr);

            let bitis = new Date(bitisStr);

            if (baslangic > bitis) return 0;

            let isGunu = 0;

            let current = new Date(baslangic);

            while (current <= bitis) {

                let day = current.getDay();

                let dateStr = current.toISOString().split('T')[0];

                if (day !== 0 && day !== 6 && !RESMI_TATILLER.includes(dateStr)) {

                    isGunu++;

                }

                current.setDate(current.getDate() + 1);

            }

            return isGunu;

        };


        const yasHesapla = (dogumTarihiStr) => {

            if(!dogumTarihiStr) return 30;

            const today = new Date();

            const birthDate = new Date(dogumTarihiStr);

            let age = today.getFullYear() - birthDate.getFullYear();

            const m = today.getMonth() - birthDate.getMonth();

            if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) age--;

            return age;

        };


        const kidemYiliHesapla = (iseBaslamaStr) => {

            if(!iseBaslamaStr) return 0;

            const today = new Date();

            const startDate = new Date(iseBaslamaStr);

            let diff = (today - startDate) / (1000 * 60 * 60 * 24 * 365.25);

            return Math.floor(diff);

        };


        const yillikIzinHakkiHesapla = (iseBaslamaStr, dogumTarihiStr) => {

            const kidem = kidemYiliHesapla(iseBaslamaStr);

            if (kidem < 1) return 0;

            let toplamHak = 0;

            for (let i = 1; i <= kidem; i++) {

                let yillikHak = 14;

                if (i >= 5 && i < 15) yillikHak = 20;

                if (i >= 15) yillikHak = 26;

                if (yas <= 18 || yas >= 50) yillikHak = Math.max(yillikHak, 20);

                toplamHak += yillikHak;

            }

            return toplamHak;

        };
